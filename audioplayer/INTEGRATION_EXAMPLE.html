Custom Playlist Example

Open player with custom tracks

Integration Code:

Method 1: Load from Artist Folder

// Automatically loads manifest.json from folder
window.RokkoPlayer.openPlayer({
    artistFolder: 'mp3/vandango',
    artistName: 'Vandango'
});

Method 2: Custom Playlist

window.RokkoPlayer.openPlayer({
    playlist: [
        {
            title: "Track Title",
            artist: "Artist Name",
            audioSrc: "path/to/audio.m4a",
            coverSrc: "path/to/cover.png"
        }
    ],
    startIndex: 0
});

Method 3: HTML Button Integration

<button onclick="window.RokkoPlayer.openPlayer({
    artistFolder: 'mp3/vandango',
    artistName: 'Vandango'
})">
    Play Vandango
</button>
